import { ChangeEvent, MutableRefObject, useCallback, useEffect, useMemo, useRef, useState } from "react";
import {
  Mic,
  Square,
  Copy,
  Download,
  Trash2,
  ArrowRightLeft,
  Upload,
  Sparkles,
  Loader2,
  AlertTriangle,
} from "lucide-react";
import { LANGUAGES } from "../constants/languages";
import { ActivePanelActions, ToastVariant } from "../types";
import { WaveformVisualizer } from "./WaveformVisualizer";

interface SpeechRecognitionAlternativeLike {
  transcript: string;
  confidence: number;
}

interface SpeechRecognitionResultLike {
  isFinal: boolean;
  length: number;
  [index: number]: SpeechRecognitionAlternativeLike;
}

interface SpeechRecognitionEventLike extends Event {
  resultIndex: number;
  results: {
    length: number;
    [index: number]: SpeechRecognitionResultLike;
  };
}

interface SpeechRecognitionErrorEventLike extends Event {
  error: string;
}

interface SpeechRecognitionInstanceLike extends EventTarget {
  continuous: boolean;
  interimResults: boolean;
  lang: string;
  start: () => void;
  stop: () => void;
  abort: () => void;
  onresult: ((event: SpeechRecognitionEventLike) => void) | null;
  onerror: ((event: SpeechRecognitionErrorEventLike) => void) | null;
  onend: (() => void) | null;
}

interface SpeechToTextPanelProps {
  transcript: string;
  onTranscriptChange: (value: string) => void;
  languageCode: string;
  onLanguageChange: (code: string) => void;
  onSendToTts: (text: string, langCode: string) => void;
  onNotify: (message: string, variant?: ToastVariant) => void;
  onSaveHistory: (entry: {
    mode: "stt";
    languageCode: string;
    languageLabel: string;
    text: string;
  }) => void;
  actionsRef?: MutableRefObject<ActivePanelActions | null>;
}

function getSpeechRecognitionConstructor():
  | (new () => SpeechRecognitionInstanceLike)
  | null {
  if (typeof window === "undefined") return null;
  const win = window as unknown as {
    SpeechRecognition?: new () => SpeechRecognitionInstanceLike;
    webkitSpeechRecognition?: new () => SpeechRecognitionInstanceLike;
  };
  return win.SpeechRecognition || win.webkitSpeechRecognition || null;
}

function formatDuration(totalSeconds: number): string {
  const mins = Math.floor(totalSeconds / 60);
  const secs = totalSeconds % 60;
  return `${String(mins).padStart(2, "0")}:${String(secs).padStart(2, "0")}`;
}

async function blobToBase64(blob: Blob): Promise<string> {
  const arrayBuffer = await blob.arrayBuffer();
  const bytes = new Uint8Array(arrayBuffer);
  let binary = "";
  const chunkSize = 0x8000;
  for (let i = 0; i < bytes.length; i += chunkSize) {
    binary += String.fromCharCode(...bytes.subarray(i, i + chunkSize));
  }
  return btoa(binary);
}

export function SpeechToTextPanel({
  transcript,
  onTranscriptChange,
  languageCode,
  onLanguageChange,
  onSendToTts,
  onNotify,
  onSaveHistory,
  actionsRef,
}: SpeechToTextPanelProps) {
  const [isRecording, setIsRecording] = useState(false);
  const [isTranscribing, setIsTranscribing] = useState(false);
  const [isDetectingLanguage, setIsDetectingLanguage] = useState(false);
  const [interimText, setInterimText] = useState("");
  const [elapsedSeconds, setElapsedSeconds] = useState(0);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [analyserNode, setAnalyserNode] = useState<AnalyserNode | null>(null);
  const [hasRecordedAudio, setHasRecordedAudio] = useState(false);

  const recognitionRef = useRef<SpeechRecognitionInstanceLike | null>(null);
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
  const mediaStreamRef = useRef<MediaStream | null>(null);
  const audioContextRef = useRef<AudioContext | null>(null);
  const audioChunksRef = useRef<Blob[]>([]);
  const lastAudioBlobRef = useRef<Blob | null>(null);
  const fileInputRef = useRef<HTMLInputElement | null>(null);
  const transcriptRef = useRef(transcript);
  const languageCodeRef = useRef(languageCode);
  const isRecordingRef = useRef(false);
  const liveTranscriptReceivedRef = useRef(false);
  const detectionTimerRef = useRef<number | null>(null);
  const autoDetectedInSessionRef = useRef<string | null>(null);
  const baseTranscriptRef = useRef<string>("");

  useEffect(() => {
    transcriptRef.current = transcript;
  }, [transcript]);

  useEffect(() => {
    languageCodeRef.current = languageCode;
  }, [languageCode]);

  const speechRecognitionSupported = useMemo(
    () => Boolean(getSpeechRecognitionConstructor()),
    []
  );

  const mediaRecordingSupported =
    typeof navigator !== "undefined" &&
    Boolean(navigator.mediaDevices && navigator.mediaDevices.getUserMedia);

  const selectedLanguageLabel = useMemo(() => {
    if (languageCode === "auto") return "Auto-detect";
    return LANGUAGES.find((l) => l.code === languageCode)?.label ?? "Hindi";
  }, [languageCode]);

  // Timer tick while recording
  useEffect(() => {
    if (!isRecording) return;
    const interval = window.setInterval(() => {
      setElapsedSeconds((prev) => prev + 1);
    }, 1000);
    return () => window.clearInterval(interval);
  }, [isRecording]);

  const cleanupAudioGraph = useCallback(() => {
    if (mediaStreamRef.current) {
      mediaStreamRef.current.getTracks().forEach((track) => track.stop());
      mediaStreamRef.current = null;
    }
    if (audioContextRef.current && audioContextRef.current.state !== "closed") {
      audioContextRef.current.close().catch(() => {});
      audioContextRef.current = null;
    }
    setAnalyserNode(null);
  }, []);

  const clearDetectionTimer = useCallback(() => {
    if (detectionTimerRef.current !== null) {
      window.clearTimeout(detectionTimerRef.current);
      detectionTimerRef.current = null;
    }
  }, []);

  useEffect(() => {
    return () => {
      isRecordingRef.current = false;
      clearDetectionTimer();
      recognitionRef.current?.abort();
      if (
        mediaRecorderRef.current &&
        mediaRecorderRef.current.state !== "inactive"
      ) {
        mediaRecorderRef.current.stop();
      }
      cleanupAudioGraph();
    };
  }, [cleanupAudioGraph, clearDetectionTimer]);

  const transcribeBlobWithServer = useCallback(
    async (
      blob: Blob,
      options?: {
        replaceAll?: boolean;
        replaceSessionOnly?: boolean;
        overrideLanguageCode?: string;
      }
    ) => {
      try {
        setIsTranscribing(true);
        const base64 = await blobToBase64(blob);
        const activeCode = options?.overrideLanguageCode || languageCodeRef.current;
        const activeLabel =
          activeCode === "auto"
            ? "Auto-detect"
            : LANGUAGES.find((l) => l.code === activeCode)?.label ?? "Hindi";

        const res = await fetch("/api/stt", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            audioBase64: base64,
            mimeType: blob.type || "audio/webm",
            languageLabel: activeLabel,
          }),
        });

        const data = await res.json();
        if (!res.ok) {
          throw new Error(data?.error || "Audio transcription failed.");
        }

        let resolvedCode = activeCode;
        let resolvedLabel = activeLabel;

        if (
          activeCode === "auto" &&
          data.detectedLanguageCode &&
          data.detectedLanguageCode !== "unknown"
        ) {
          const matched = LANGUAGES.find(
            (l) => l.code === data.detectedLanguageCode
          );
          if (matched) {
            resolvedCode = matched.code;
            resolvedLabel = matched.label;
            onLanguageChange(matched.code);
            onNotify(`Detected language: ${matched.label}`, "info");
          }
        }

        const newText = (data.transcript || "").trim();
        if (!newText) {
          onNotify("No speech was detected in the recording.", "info");
          return;
        }

        let updated = newText;
        if (options?.replaceAll) {
          updated = newText;
        } else if (options?.replaceSessionOnly) {
          const base = baseTranscriptRef.current;
          updated = base ? `${base} ${newText}` : newText;
        } else {
          const existing = transcriptRef.current.trim();
          updated = existing ? `${existing} ${newText}` : newText;
        }

        onTranscriptChange(updated);
        onSaveHistory({
          mode: "stt",
          languageCode: resolvedCode === "auto" ? "hi-IN" : resolvedCode,
          languageLabel: resolvedLabel,
          text: updated,
        });
        onNotify("Transcription complete", "success");
      } catch (err) {
        const msg =
          err instanceof Error ? err.message : "Failed to transcribe audio.";
        setErrorMessage(msg);
        onNotify(msg, "error");
      } finally {
        setIsTranscribing(false);
      }
    },
    [onLanguageChange, onNotify, onSaveHistory, onTranscriptChange]
  );

  const stopRecording = useCallback(() => {
    isRecordingRef.current = false;
    clearDetectionTimer();
    setIsRecording(false);
    setIsDetectingLanguage(false);
    setInterimText("");

    if (recognitionRef.current) {
      try {
        recognitionRef.current.stop();
      } catch {
        // Ignore stop errors if already ended
      }
      recognitionRef.current = null;
    }

    if (
      mediaRecorderRef.current &&
      mediaRecorderRef.current.state !== "inactive"
    ) {
      mediaRecorderRef.current.stop();
    } else {
      cleanupAudioGraph();
    }
  }, [cleanupAudioGraph, clearDetectionTimer]);

  const startWebSpeechRecognition = useCallback(
    (targetLang: string) => {
      const SpeechRec = getSpeechRecognitionConstructor();
      if (!SpeechRec || !isRecordingRef.current) return;

      if (recognitionRef.current) {
        try {
          recognitionRef.current.abort();
        } catch {
          // Ignore
        }
        recognitionRef.current = null;
      }

      const recognition = new SpeechRec();
      recognition.continuous = true;
      recognition.interimResults = true;
      recognition.lang = targetLang;

      recognition.onresult = (event) => {
        let finalizedChunk = "";
        let interimChunk = "";

        for (let i = event.resultIndex; i < event.results.length; i++) {
          const result = event.results[i];
          const piece = result[0]?.transcript || "";
          if (result.isFinal) {
            finalizedChunk += piece;
          } else {
            interimChunk += piece;
          }
        }

        if (finalizedChunk.trim()) {
          liveTranscriptReceivedRef.current = true;
          const current = transcriptRef.current.trim();
          const merged = current
            ? `${current} ${finalizedChunk.trim()}`
            : finalizedChunk.trim();
          onTranscriptChange(merged);
        }

        if (interimChunk.trim()) {
          liveTranscriptReceivedRef.current = true;
        }
        setInterimText(interimChunk);
      };

      recognition.onerror = (event) => {
        if (event.error === "not-allowed" || event.error === "service-not-allowed") {
          setErrorMessage(
            "Microphone permission was denied. Please allow microphone access in your browser settings."
          );
          stopRecording();
        }
      };

      recognition.onend = () => {
        if (isRecordingRef.current && recognitionRef.current === recognition) {
          try {
            recognition.start();
          } catch {
            // Ignore restart error if stopped concurrently
          }
        }
      };

      recognitionRef.current = recognition;
      try {
        recognition.start();
      } catch {
        // Ignore if already started
      }
    },
    [onTranscriptChange, stopRecording]
  );

  const scheduleAutoLanguageDetection = useCallback(
    (mimeType: string, attempt = 1) => {
      clearDetectionTimer();
      detectionTimerRef.current = window.setTimeout(async () => {
        if (!isRecordingRef.current || audioChunksRef.current.length === 0) {
          return;
        }

        try {
          const sampleBlob = new Blob(audioChunksRef.current, { type: mimeType });
          const base64 = await blobToBase64(sampleBlob);

          const res = await fetch("/api/detect-language", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
              audioBase64: base64,
              mimeType,
            }),
          });

          const data = await res.json();
          if (!isRecordingRef.current) return;

          const detectedCode = data?.languageCode;
          const matchedLang = LANGUAGES.find((l) => l.code === detectedCode);

          if (res.ok && matchedLang) {
            autoDetectedInSessionRef.current = matchedLang.code;
            setIsDetectingLanguage(false);
            onLanguageChange(matchedLang.code);
            onNotify(`Language detected: ${matchedLang.label}`, "success");

            const initialText = (data?.transcript || "").trim();
            if (initialText) {
              const base = baseTranscriptRef.current;
              onTranscriptChange(base ? `${base} ${initialText}` : initialText);
            }

            if (speechRecognitionSupported) {
              startWebSpeechRecognition(matchedLang.code);
            }
            return;
          }

          if (attempt < 2 && isRecordingRef.current) {
            scheduleAutoLanguageDetection(mimeType, attempt + 1);
          } else {
            setIsDetectingLanguage(false);
          }
        } catch {
          if (isRecordingRef.current) {
            setIsDetectingLanguage(false);
          }
        }
      }, 3000);
    },
    [
      clearDetectionTimer,
      onLanguageChange,
      onNotify,
      onTranscriptChange,
      speechRecognitionSupported,
      startWebSpeechRecognition,
    ]
  );

  const startRecording = async () => {
    setErrorMessage(null);
    setElapsedSeconds(0);
    setInterimText("");
    liveTranscriptReceivedRef.current = false;
    autoDetectedInSessionRef.current = null;
    baseTranscriptRef.current = transcriptRef.current.trim();

    if (!mediaRecordingSupported && !speechRecognitionSupported) {
      setErrorMessage(
        "Your browser does not support microphone capture or Web Speech Recognition. You can upload an audio file below instead."
      );
      return;
    }

    const startingInAutoMode = languageCodeRef.current === "auto";

    try {
      let stream: MediaStream | null = null;
      let recorderMime = "audio/webm";

      if (mediaRecordingSupported) {
        stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        mediaStreamRef.current = stream;

        // Real-time audio analyser for live waveform
        const AudioCtx =
          window.AudioContext ||
          (window as unknown as { webkitAudioContext: typeof AudioContext })
            .webkitAudioContext;
        if (AudioCtx) {
          const ctx = new AudioCtx();
          const source = ctx.createMediaStreamSource(stream);
          const analyser = ctx.createAnalyser();
          analyser.fftSize = 128;
          source.connect(analyser);
          audioContextRef.current = ctx;
          setAnalyserNode(analyser);
        }

        // Record audio in parallel for auto-detection and AI transcription
        audioChunksRef.current = [];
        const preferredMime = MediaRecorder.isTypeSupported("audio/webm;codecs=opus")
          ? "audio/webm;codecs=opus"
          : MediaRecorder.isTypeSupported("audio/webm")
          ? "audio/webm"
          : "";
        const recorder = preferredMime
          ? new MediaRecorder(stream, { mimeType: preferredMime })
          : new MediaRecorder(stream);
        recorderMime = recorder.mimeType || preferredMime || "audio/webm";

        recorder.ondataavailable = (ev) => {
          if (ev.data.size > 0) {
            audioChunksRef.current.push(ev.data);
          }
        };

        recorder.onstop = async () => {
          clearDetectionTimer();
          setIsDetectingLanguage(false);
          const mimeType = recorder.mimeType || recorderMime || "audio/webm";
          const recordedBlob = new Blob(audioChunksRef.current, { type: mimeType });
          cleanupAudioGraph();

          if (recordedBlob.size > 0) {
            lastAudioBlobRef.current = recordedBlob;
            setHasRecordedAudio(true);

            const activeCode = languageCodeRef.current;
            const activeLabel =
              activeCode === "auto"
                ? "Auto-detect"
                : LANGUAGES.find((l) => l.code === activeCode)?.label ?? "Hindi";

            // Transcribe full clip via Gemini STT if live Web Speech wasn't active,
            // stopped before auto-detection finished, or received no live output.
            if (
              !speechRecognitionSupported ||
              activeCode === "auto" ||
              !liveTranscriptReceivedRef.current
            ) {
              await transcribeBlobWithServer(recordedBlob, {
                replaceSessionOnly: Boolean(autoDetectedInSessionRef.current),
                overrideLanguageCode: autoDetectedInSessionRef.current || activeCode,
              });
            } else if (transcriptRef.current.trim()) {
              onSaveHistory({
                mode: "stt",
                languageCode: activeCode,
                languageLabel: activeLabel,
                text: transcriptRef.current.trim(),
              });
            }
          }
        };

        mediaRecorderRef.current = recorder;
        recorder.start(350);
      }

      isRecordingRef.current = true;
      setIsRecording(true);

      if (startingInAutoMode) {
        if (mediaRecordingSupported) {
          setIsDetectingLanguage(true);
          scheduleAutoLanguageDetection(recorderMime, 1);
        }
      } else if (speechRecognitionSupported) {
        startWebSpeechRecognition(languageCodeRef.current);
      }
    } catch (err) {
      clearDetectionTimer();
      cleanupAudioGraph();
      isRecordingRef.current = false;
      setIsRecording(false);
      setIsDetectingLanguage(false);
      const msg =
        err instanceof Error && err.name === "NotAllowedError"
          ? "Microphone access was denied. Please grant microphone permission and try again."
          : "Unable to access microphone. Check your audio input device or upload an audio file.";
      setErrorMessage(msg);
      onNotify(msg, "error");
    }
  };

  const handleCopy = async () => {
    const content = transcript.trim();
    if (!content) {
      onNotify("Nothing to copy yet.", "info");
      return;
    }
    try {
      await navigator.clipboard.writeText(content);
      onNotify("Transcript copied to clipboard", "success");
    } catch {
      onNotify("Could not access clipboard. Select and copy manually.", "error");
    }
  };

  const handleDownloadTxt = () => {
    const content = transcript.trim();
    if (!content) {
      onNotify("Nothing to download yet.", "info");
      return;
    }
    const blob = new Blob([content], { type: "text/plain;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");
    link.href = url;
    link.download = `vaaniai-transcript-${Date.now()}.txt`;
    document.body.appendChild(link);
    link.click();
    link.remove();
    URL.revokeObjectURL(url);
    onNotify("Transcript downloaded (.txt)", "success");
  };

  const handleFileUpload = async (e: ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    e.target.value = "";
    if (!file) return;

    if (file.size > 15 * 1024 * 1024) {
      onNotify("Please upload an audio file smaller than 15 MB.", "error");
      return;
    }

    setErrorMessage(null);
    await transcribeBlobWithServer(file);
  };

  useEffect(() => {
    if (!actionsRef) return;
    actionsRef.current = {
      primaryAction: () => {
        if (isTranscribing) return;
        if (isRecording) {
          stopRecording();
        } else {
          void startRecording();
        }
      },
      stopAction: () => {
        if (isRecording) {
          stopRecording();
        }
      },
      downloadAction: () => {
        handleDownloadTxt();
      },
      copyAction: () => {
        void handleCopy();
      },
    };
    return () => {
      actionsRef.current = null;
    };
  });

  const wordCount = transcript.trim()
    ? transcript.trim().split(/\s+/).length
    : 0;

  return (
    <div className="space-y-6">
      {/* Recognition Controls Row */}
      <div className="flex flex-wrap items-center justify-between gap-4">
        <div className="w-full sm:w-72">
          <label
            htmlFor="stt-language"
            className="block text-xs font-medium text-slate-300 mb-1.5"
          >
            Spoken Language
          </label>
          <select
            id="stt-language"
            value={languageCode}
            disabled={isRecording}
            onChange={(e) => onLanguageChange(e.target.value)}
            className="w-full min-h-[44px] rounded-xl bg-slate-950/75 border border-white/10 focus:border-indigo-400/60 focus:ring-2 focus:ring-indigo-500/25 px-3.5 py-2.5 text-sm text-slate-100 outline-none disabled:opacity-60"
          >
            <option value="auto" className="bg-[#0E1222]">
              Auto-detect Language (Studio AI)
            </option>
            {LANGUAGES.map((lang) => (
              <option key={lang.code} value={lang.code} className="bg-[#0E1222]">
                {lang.label} — {lang.nativeName}
              </option>
            ))}
          </select>
        </div>

        <div className="flex flex-wrap items-center gap-2.5 sm:pt-6">
          <input
            ref={fileInputRef}
            type="file"
            accept="audio/*"
            onChange={handleFileUpload}
            className="hidden"
          />
          <button
            type="button"
            onClick={() => fileInputRef.current?.click()}
            disabled={isRecording || isTranscribing}
            className="inline-flex items-center gap-2 min-h-[44px] px-3.5 py-2 rounded-xl bg-white/5 hover:bg-white/10 border border-white/10 disabled:opacity-50 text-xs font-medium text-slate-200 transition-colors whitespace-nowrap shrink-0"
          >
            <Upload className="w-3.5 h-3.5" />
            Upload Audio
          </button>

          {hasRecordedAudio && !isRecording && (
            <button
              type="button"
              onClick={() => {
                if (lastAudioBlobRef.current) {
                  transcribeBlobWithServer(lastAudioBlobRef.current, {
                    replaceAll: true,
                  });
                }
              }}
              disabled={isTranscribing}
              className="inline-flex items-center gap-2 min-h-[44px] px-3.5 py-2 rounded-xl bg-indigo-500/15 hover:bg-indigo-500/25 border border-indigo-400/30 disabled:opacity-50 text-xs font-medium text-indigo-100 transition-colors whitespace-nowrap shrink-0"
            >
              <Sparkles className="w-3.5 h-3.5" />
              Refine with AI
            </button>
          )}
        </div>
      </div>

      {/* Browser Notice if SpeechRecognition is absent */}
      {!speechRecognitionSupported && (
        <div
          role="status"
          className="flex items-start gap-3 rounded-xl bg-indigo-500/10 border border-indigo-400/25 px-4 py-3 text-sm text-indigo-100"
        >
          <Sparkles className="w-4 h-4 text-indigo-400 shrink-0 mt-0.5" />
          <div>
            <p className="font-medium">Using Studio AI Recording Mode</p>
            <p className="text-xs text-indigo-200/80 mt-0.5">
              Live browser speech streaming is unavailable in this browser. Click the microphone to record—your speech will transcribe automatically as soon as you stop.
            </p>
          </div>
        </div>
      )}

      {errorMessage && (
        <div
          role="alert"
          className="flex items-start gap-3 rounded-xl bg-rose-500/10 border border-rose-400/25 px-4 py-3 text-sm text-rose-100"
        >
          <AlertTriangle className="w-4 h-4 text-rose-400 shrink-0 mt-0.5" />
          <p className="leading-snug">{errorMessage}</p>
        </div>
      )}

      {/* Focal Microphone Stage + Live Waveform */}
      <div className="rounded-2xl bg-slate-950/60 border border-white/10 p-6 flex flex-col items-center text-center gap-5">
        <div className="relative flex items-center justify-center">
          {isRecording && (
            <>
              <span
                className="absolute w-28 h-28 rounded-full bg-indigo-500/20 animate-ping"
                aria-hidden="true"
              />
              <span
                className="absolute w-24 h-24 rounded-full bg-violet-500/25 animate-pulse"
                aria-hidden="true"
              />
            </>
          )}
          <button
            type="button"
            onClick={isRecording ? stopRecording : startRecording}
            disabled={isTranscribing}
            aria-label={isRecording ? "Stop recording" : "Start recording"}
            className={`relative z-10 w-20 h-20 rounded-full flex items-center justify-center transition-transform duration-200 focus-visible:outline-none focus-visible:ring-4 focus-visible:ring-indigo-400/40 active:scale-95 ${
              isRecording
                ? "bg-rose-600 hover:bg-rose-500 text-white shadow-xl shadow-rose-600/35"
                : "bg-gradient-to-br from-indigo-500 to-violet-600 hover:from-indigo-400 hover:to-violet-500 text-white shadow-xl shadow-indigo-600/35"
            } disabled:opacity-60`}
          >
            {isTranscribing ? (
              <Loader2 className="w-8 h-8 animate-spin" />
            ) : isRecording ? (
              <Square className="w-7 h-7 fill-current" />
            ) : (
              <Mic className="w-8 h-8" />
            )}
          </button>
        </div>

        <div className="space-y-1">
          <div className="flex items-center justify-center gap-3">
            <span className="text-sm font-medium text-slate-200">
              {isTranscribing
                ? "Transcribing audio..."
                : isRecording
                ? isDetectingLanguage
                  ? "Listening — identifying spoken language..."
                  : languageCode === "auto" || !speechRecognitionSupported
                  ? "Recording audio — click stop to transcribe"
                  : "Listening live — speak clearly"
                : "Tap microphone to start speaking"}
            </span>
            <span className="font-mono text-sm text-indigo-300 font-semibold">
              {formatDuration(elapsedSeconds)}
            </span>
          </div>
          <p className="text-xs text-slate-400">
            Language: {selectedLanguageLabel} · Press{" "}
            <kbd className="px-1.5 py-0.5 rounded bg-white/10 text-slate-300 font-mono text-[11px]">
              Ctrl/⌘ + Enter
            </kbd>{" "}
            to {isRecording ? "stop" : "transcribe"}
          </p>
        </div>

        <WaveformVisualizer
          active={isRecording}
          analyser={analyserNode}
          simulated={isRecording && !analyserNode}
          accentColor="violet"
          label="Live microphone input waveform"
        />
      </div>

      {/* Live Transcript Output Box */}
      <div className="space-y-2">
        <div className="flex items-center justify-between text-xs text-slate-400">
          <label htmlFor="stt-transcript" className="font-medium text-slate-300">
            Transcript
          </label>
          <div className="flex items-center gap-2 font-mono">
            <span>{wordCount.toLocaleString()} words</span>
            <span aria-hidden="true">·</span>
            <span>{transcript.length.toLocaleString()} chars</span>
          </div>
        </div>

        <textarea
          id="stt-transcript"
          value={transcript}
          onChange={(e) => onTranscriptChange(e.target.value)}
          rows={6}
          placeholder="Your transcribed speech will appear here in real time. You can also edit the text directly before copying or downloading."
          className="w-full rounded-xl bg-slate-950/65 border border-white/10 focus:border-indigo-400/60 focus:ring-2 focus:ring-indigo-500/25 px-4 py-3.5 text-base text-slate-100 placeholder:text-slate-500 resize-y min-h-[168px] leading-relaxed outline-none transition-colors"
        />

        {interimText && (
          <div className="rounded-lg bg-indigo-500/10 border border-indigo-400/20 px-3.5 py-2 text-sm text-indigo-200">
            <span className="text-xs font-medium text-indigo-300 mr-2">
              Hearing:
            </span>
            {interimText}
          </div>
        )}
      </div>

      {/* Transcript Actions */}
      <div className="flex flex-wrap items-center justify-between gap-3 pt-2 border-t border-white/10">
        <div className="flex flex-wrap items-center gap-2.5">
          <button
            type="button"
            onClick={handleCopy}
            className="inline-flex items-center justify-center gap-2 min-h-[44px] px-4 py-2.5 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-sm font-semibold shadow-lg shadow-indigo-600/20 transition-colors whitespace-nowrap shrink-0"
          >
            <Copy className="w-4 h-4" />
            Copy Text
          </button>
          <button
            type="button"
            onClick={handleDownloadTxt}
            className="inline-flex items-center justify-center gap-2 min-h-[44px] px-4 py-2.5 rounded-xl bg-white/5 hover:bg-white/10 border border-white/10 text-slate-100 text-sm font-medium transition-colors whitespace-nowrap shrink-0"
          >
            <Download className="w-4 h-4" />
            Download .TXT
          </button>
        </div>

        <div className="flex flex-wrap items-center gap-2.5">
          {transcript.trim().length > 0 && (
            <>
              <button
                type="button"
                onClick={() =>
                  onSendToTts(
                    transcript.trim(),
                    languageCode === "auto" ? "hi-IN" : languageCode
                  )
                }
                className="inline-flex items-center justify-center gap-2 min-h-[44px] px-4 py-2.5 rounded-xl bg-violet-500/15 hover:bg-violet-500/25 border border-violet-400/30 text-violet-100 text-sm font-medium transition-colors whitespace-nowrap shrink-0"
              >
                <ArrowRightLeft className="w-4 h-4" />
                Speak in TTS
              </button>
              <button
                type="button"
                onClick={() => {
                  onTranscriptChange("");
                  setInterimText("");
                  setElapsedSeconds(0);
                  setHasRecordedAudio(false);
                  lastAudioBlobRef.current = null;
                }}
                className="inline-flex items-center justify-center gap-1.5 min-h-[44px] px-3.5 py-2.5 rounded-xl text-slate-300 hover:text-rose-200 hover:bg-rose-500/15 border border-white/10 text-sm font-medium transition-colors whitespace-nowrap shrink-0"
              >
                <Trash2 className="w-4 h-4" />
                Clear
              </button>
            </>
          )}
        </div>
      </div>
    </div>
  );
}
