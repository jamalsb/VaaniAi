import { X, Trash2, ArrowUpRight, Copy, Volume2, Mic, Download } from "lucide-react";
import { HistoryEntry } from "../types";

interface HistoryDrawerProps {
  open: boolean;
  entries: HistoryEntry[];
  onClose: () => void;
  onRestore: (entry: HistoryEntry) => void;
  onCopy: (text: string) => void;
  onDelete: (id: string) => void;
  onExportJson: () => void;
  onClearAll: () => void;
}

export function HistoryDrawer({
  open,
  entries,
  onClose,
  onRestore,
  onCopy,
  onDelete,
  onExportJson,
  onClearAll,
}: HistoryDrawerProps) {
  if (!open) return null;

  const formatTimestamp = (iso: string) => {
    try {
      return new Intl.DateTimeFormat(undefined, {
        month: "short",
        day: "numeric",
        hour: "2-digit",
        minute: "2-digit",
      }).format(new Date(iso));
    } catch {
      return "";
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex justify-end">
      <div
        className="fixed inset-0 bg-black/60 backdrop-blur-sm transition-opacity"
        onClick={onClose}
        aria-hidden="true"
      />
      <aside
        role="dialog"
        aria-label="Conversion History"
        className="relative z-10 w-full max-w-md h-full bg-[#0E1222] border-l border-white/10 flex flex-col shadow-2xl"
      >
        <div className="flex items-center justify-between px-6 py-5 border-b border-white/10">
          <div>
            <h2 className="text-lg font-semibold text-slate-100 font-display">
              Recent Activity
            </h2>
            <p className="text-xs text-slate-400 mt-0.5">
              Saved locally in your browser
            </p>
          </div>
          <div className="flex items-center gap-2">
            {entries.length > 0 && (
              <button
                type="button"
                onClick={onClearAll}
                className="px-3 py-1.5 text-xs font-medium text-slate-300 hover:text-rose-300 hover:bg-rose-500/10 rounded-lg transition-colors whitespace-nowrap shrink-0"
              >
                Clear All
              </button>
            )}
            <button
              type="button"
              onClick={onClose}
              className="p-2 rounded-lg text-slate-400 hover:text-slate-100 hover:bg-white/10 transition-colors"
              aria-label="Close history"
            >
              <X className="w-4 h-4" />
            </button>
          </div>
        </div>

        <div className="flex-1 overflow-y-auto p-5 space-y-3">
          {entries.length === 0 ? (
            <div className="h-full flex flex-col items-center justify-center text-center px-6 py-12">
              <div className="w-12 h-12 rounded-2xl bg-indigo-500/10 border border-indigo-400/20 flex items-center justify-center mb-4">
                <Volume2 className="w-5 h-5 text-indigo-400" />
              </div>
              <p className="text-sm font-medium text-slate-200">
                No saved conversions yet
              </p>
              <p className="text-xs text-slate-400 mt-1.5 leading-relaxed">
                Generated speech and voice transcripts will appear here automatically so you can revisit them anytime.
              </p>
            </div>
          ) : (
            entries.map((entry) => (
              <div
                key={entry.id}
                className="rounded-xl bg-white/[0.03] border border-white/10 p-4 hover:border-indigo-400/30 transition-colors"
              >
                <div className="flex items-center justify-between gap-2 text-xs text-slate-400 mb-2">
                  <div className="flex items-center gap-2 min-w-0">
                    {entry.mode === "tts" ? (
                      <Volume2 className="w-3.5 h-3.5 text-indigo-400 shrink-0" />
                    ) : (
                      <Mic className="w-3.5 h-3.5 text-violet-400 shrink-0" />
                    )}
                    <span className="font-medium text-slate-300 truncate">
                      {entry.mode === "tts" ? "Text to Speech" : "Speech to Text"}
                    </span>
                    <span aria-hidden="true">·</span>
                    <span className="truncate">{entry.languageLabel}</span>
                  </div>
                  <span className="font-mono text-[11px] text-slate-500 shrink-0">
                    {formatTimestamp(entry.createdAt)}
                  </span>
                </div>

                <p className="text-sm text-slate-200 line-clamp-3 leading-relaxed">
                  {entry.text}
                </p>

                <div className="flex items-center justify-end gap-2 mt-3 pt-3 border-t border-white/5">
                  <button
                    type="button"
                    onClick={() => onCopy(entry.text)}
                    className="inline-flex items-center gap-1.5 px-2.5 py-1.5 text-xs font-medium text-slate-300 hover:text-white hover:bg-white/10 rounded-lg transition-colors whitespace-nowrap shrink-0"
                  >
                    <Copy className="w-3.5 h-3.5" />
                    Copy
                  </button>
                  <button
                    type="button"
                    onClick={() => onDelete(entry.id)}
                    className="inline-flex items-center gap-1.5 px-2.5 py-1.5 text-xs font-medium text-slate-400 hover:text-rose-300 hover:bg-rose-500/10 rounded-lg transition-colors whitespace-nowrap shrink-0"
                    aria-label="Delete entry"
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                  </button>
                  <button
                    type="button"
                    onClick={() => {
                      onRestore(entry);
                      onClose();
                    }}
                    className="inline-flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium text-indigo-200 bg-indigo-500/20 hover:bg-indigo-500/30 rounded-lg transition-colors whitespace-nowrap shrink-0"
                  >
                    Open
                    <ArrowUpRight className="w-3.5 h-3.5" />
                  </button>
                </div>
              </div>
            ))
          )}
        </div>

        {entries.length > 0 && (
          <div className="px-6 py-4 border-t border-white/10 bg-slate-950/50 flex items-center justify-between gap-4">
            <span className="text-xs text-slate-400 font-mono">
              {entries.length} {entries.length === 1 ? "entry" : "entries"} saved
            </span>
            <button
              type="button"
              onClick={onExportJson}
              className="inline-flex items-center justify-center gap-2 min-h-[40px] px-4 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-xs font-semibold shadow-md shadow-indigo-600/25 transition-colors whitespace-nowrap shrink-0"
            >
              <Download className="w-3.5 h-3.5" />
              Export JSON
            </button>
          </div>
        )}
      </aside>
    </div>
  );
}
